Google Sheets / Apps Script Sync Endpoint

Telemetry & ledger records ko Google Sheet mein bhejne ke liye Web App URL save karein.

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var payload = JSON.parse(e.postData.contents);
  sheet.appendRow([
    new Date().toISOString(),
    payload.type || "event",
    payload.qrId || "",
    payload.technician ? payload.technician.name : "",
    payload.technician ? payload.technician.mobile : "",
    payload.trade || "",
    JSON.stringify(payload.meta || {})
  ]);
  return ContentService.createTextOutput('{"status":"ok"}')
    .setMimeType(ContentService.MimeType.JSON);
}
